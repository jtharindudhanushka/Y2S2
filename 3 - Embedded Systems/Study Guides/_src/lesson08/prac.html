Practical question: "Serial command console" (exam style)

Original practice question combining Exercise 7 (USART, LM35) with interrupts and a counted event.

ScenarioA bench instrument is controlled from a PC terminal over the USB serial link at 19 200 baud (8 data bits, no parity, one stop bit). The PC sends single characters:
• 'L' toggles an LED on the board;
• 'T' makes the board reply with the room temperature in whole °C (LM35) followed by a new line;
• 'C' makes the board reply with the number of times a pushbutton has been pressed since reset, followed by a new line;
• any other character gets the reply ? and a new line.
The button presses must never be missed, even while a reply is being sent.
  1. Calculate UBRR and the baud rate error for 19 200 baud at 16 MHz. How long does the reply "27\n" take on the wire?
  2. Give the connection list.
  3. Write the firmware in C (registers, labelled constants).
  4. Write the firmware with the Arduino wrapper functions.
  5. Explain which pins you must keep free and why.
Solution

1. Baud rate

UBRR = 16 000 000 / (16 × 19 200) − 1 = 51.08, so UBRR = 51. Actual = 16 000 000 / (16 × 52) = 19 230.8 baud. Error = (19 230.8 − 19 200) / 19 200 = +0.16 %, fine. The reply "27\n" has 3 characters × 10 bits = 30 bits, so 30 / 19 200 = 1.56 ms.

3. C firmware

#include <avr/io.h>
#include <avr/interrupt.h>
#include <stdlib.h>                                     // itoa()

volatile unsigned int presses = 0;

static void txChar(char c)
{
    while (!(UCSR0A & (1<<UDRE0)));
    UDR0 = c;
}
static void txString(const char *s) { while (*s) txChar(*s++); }
static void txNumber(unsigned int n)
{
    char buf[6];
    itoa(n, buf, 10);
    txString(buf);
    txChar('\n');
}

int main(void)
{
    DDRB  |=  (1<<DDB0);                               // LED output
    DDRD  &= ~(1<<DDD2);   PORTD |= (1<<PORTD2);       // button input + pull-up
    EICRA |=  (1<<ISC01);    EICRA &= ~(1<<ISC00);      // INT0 falling edge
    EIMSK |=  (1<<INT0);

    ADMUX  = (1<<REFS0);                                // AVcc, ADC0
    ADCSRA = (1<<ADEN)|(1<<ADPS2)|(1<<ADPS1)|(1<<ADPS0);

    UBRR0  = 16000000UL / (16 * 19200UL) - 1;           // 51
    UCSR0B &= ~(1<<UCSZ02);
    UCSR0C |= (1<<UCSZ01) | (1<<UCSZ00);               // 8 bit
    UCSR0B |= (1<<RXEN0) | (1<<TXEN0);                 // enable RX and TX
    sei();

    while (1)
    {
        while (!(UCSR0A & (1<<RXC0)));                // wait for a command
        char cmd = UDR0;
        if (cmd == 'L')      PORTB ^= (1<<PORTB0);
        else if (cmd == 'T')
        {
            ADCSRA |= (1<<ADSC);
            while (ADCSRA & (1<<ADSC));
            txNumber((unsigned int)(ADCW * 5.0 / 1024 * 100));
        }
        else if (cmd == 'C') txNumber(presses);
        else { txChar('?'); txChar('\n'); }
    }
}

ISR(INT0_vect) { presses++; }                           // never missed: counted by the interrupt

4. Arduino firmware

volatile unsigned int presses = 0;
void count() { presses++; }
void setup() {
  Serial.begin(19200);
  pinMode(8, OUTPUT);
  pinMode(2, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(2), count, FALLING);
}
void loop() {
  if (Serial.available()) {
    char cmd = Serial.read();
    if (cmd == 'L')      digitalWrite(8, !digitalRead(8));
    else if (cmd == 'T') Serial.println((int)(analogRead(A0) * 5.0 / 1024 * 100));
    else if (cmd == 'C') Serial.println(presses);
    else                 Serial.println('?');
  }
}

5. Pins to keep free

Arduino pins 0 and 1 (PD0 = RXD, PD1 = TXD) belong to the USART and the USB interface; connecting anything there interferes with communication and with uploading the firmware.

Marking hintsUBRR with the UL suffix and the error shown; RXEN/TXEN and 8-bit set; polling RXC and UDRE correctly; itoa or println for numbers; button on INT0 with ISR and volatile; Rx/Tx pins kept free; terminal baud rate set to 19 200.
