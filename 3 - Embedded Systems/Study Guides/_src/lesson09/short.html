// ADC read          ADMUX=(1<<REFS0); ADCSRA=(1<<ADEN)|7; ADCSRA|=(1<<ADSC); while(ADCSRA&(1<<ADSC)); v=ADCW;

// PWM Timer0 /64    DDRD|=(1<<DDD6); TCCR0A=(1<<COM0A1)|(1<<WGM01)|(1<<WGM00); TCCR0B=(1<<CS01)|(1<<CS00); OCR0A=v;
// USART 9600        UBRR0=103; UCSR0C|=(1<<UCSZ01)|(1<<UCSZ00); UCSR0B|=(1<<RXEN0)|(1<<TXEN0);
